LOAD PROJ_ID, JML_STATUS_CD, OWNER_POSITION_CD, ApplyMap('ProjectName_Map', PROJ_ID, PROJ_ID) AS [Project Name], [ICN/Key Op Link] AS [ICN KO], IF(Len(Trim([JML Document Number])) = 0, 'PENDING_ACT_' & MATL_SA_ID, [JML Document Number]) AS [JML Doc #], [JML NIIN] AS [NIIN or Part Number], [JML Noun Name] AS [Part Name], Date([Matl Status - ETD]) AS [EDD], ApplyMap('COG_MAP', [JML NIIN], 'N/A') AS [COG Code], Autonumber(PROJ_ID & '|' & [ICN/Key Op Link]) AS [ProjectID ICNKO Key] FROM [lib://QVD-JRMC-AIM/MRQT.qvd] (qvd) WHERE Exists(PROJ_ID, PROJ_ID) AND (Len(Trim([JML Document Number])) > 0 OR (JML_STATUS_CD = 'ACT' AND Match(OWNER_POSITION_CD, 'YCSB', 'YAPM', 'YPCD'))) AND NOT WildMatch(Upper([JML Noun Name]), '*E-MALL*', '*EMALL*', '*RENTAL*', '*RENT,*', 'ANODIZE COATING', 'ANODIZED COATING');

LEFT JOIN (FinalReportData) LOAD [ProjectID ICNKO Key], [SKED Start], [Total DU], [Resolved Key Event ID], [Event Name], [KEY EVENT Date], CU_PHASE_SA_ID RESIDENT [Phase_Event_Agg]; DROP TABLE [Phase_Event_Agg];
LEFT JOIN (FinalReportData) LOAD CU_PHASE_SA_ID, [System], [Material Type], [Key Event], [Job Summary Title] RESIDENT JcnPhaseBridge; DROP TABLE JcnPhaseBridge;
LEFT JOIN (FinalReportData) LOAD DISTINCT PROJ_ID, [SA00 Date] RESIDENT SA00_Date_Temp; DROP TABLE SA00_Date_Temp;
LEFT JOIN (FinalReportData) LOAD [Document Number] AS [JML Doc #], [QTY Due], [NEED Date], [Unit of Issue], [Actual Status Date] RESIDENT MAT_Latest_Status; DROP TABLE MAT_Latest_Status;
LEFT JOIN (FinalReportData) LOAD [Document Number] AS [JML Doc #], [Document Cut Number] RESIDENT BridgeTable; DROP TABLE BridgeTable;
LEFT JOIN (FinalReportData) LOAD [Document Cut Number], [Status], [Parsed Status Block], [Parsed ESD], [Parsed KTR] RESIDENT DocDesc_Consolidated; DROP TABLE DocDesc_Consolidated;

DROP TABLE Projects;

Final_Table:
LOAD 
    PROJ_ID, JML_STATUS_CD, OWNER_POSITION_CD, [Project Name], [Project Display Name], [Active Project Filter], [Project Type], [Cohort Type],
    [ICN KO], [JML Doc #], [NIIN or Part Number], [Part Name], [EDD], [COG Code], [ProjectID ICNKO Key], [SKED Start], [Total DU], [Resolved Key Event ID], [Event Name], [KEY EVENT Date], CU_PHASE_SA_ID, [System], [Material Type], [Key Event], [Job Summary Title],
    [SA00 Date], [CA00 Date], [A-600 Date], [QTY Due], [NEED Date], [Unit of Issue], [Document Cut Number], [Status], [Parsed Status Block], [Parsed ESD], [Parsed KTR], [Actual Status Date], 
    
    [Unified Action Date], 
    [Receipt_EAD Relative Days from A-600],
    RangeMax(0, Ceil(([Receipt_EAD Relative Days from A-600]) / 7)) AS [Receipt_EAD Relative Week from A-600],
    
    IF([Report Status] = 'Outstanding' AND IsNum([SA00 Date]) AND [Estimated Arrival Date] > [SA00 Date], 1, 0) AS [Late to SA00 Flag],
    IF([Report Status] = 'Outstanding' AND IsNum([A-30 Date]) AND [Estimated Arrival Date] > [A-30 Date], 1, 0) AS [Late to A-30 Flag],
    IF([Report Status] = 'Outstanding' AND IsNum([Project Need Date]) AND [Estimated Arrival Date] > [Project Need Date], 1, 0) AS [Late to PND Flag],
    IF([Report Status] = 'Outstanding' AND NOT ([Estimated Arrival Date] > [SA00 Date] OR [Estimated Arrival Date] > [A-30 Date] OR [Estimated Arrival Date] > [Project Need Date]), 'On Track', Null()) AS [Remarks],
    IF(IsNum([Order Date]) AND IsNum([SA00 Date]) AND [Order Date] <= [SA00 Date], 1, 0) AS [Ordered Pre-SA00 Flag],
    IF(IsNum([Order Date]) AND IsNum([CA00 Date]) AND [Order Date] <= [CA00 Date], 1, 0) AS [Ordered Pre-CA00 Flag],
    IF(JML_STATUS_CD = 'ACT' AND Match(OWNER_POSITION_CD, 'YCSB', 'YAPM', 'YPCD') AND WildMatch([JML Doc #], 'PENDING_ACT_*'), 1, 0) AS [Is In Process ACT Flag],

    [Order Date], [Days from A-600], [Relative Week from A-600], [Report Status], [A-30 Date], [Project Need Date], [Estimated Arrival Date];

// PRECEDING LOAD
LOAD 
    *,
    // ROTN LOGIC (Riding On Time Now) applied directly to the Unified timeline
    IF(Index(Upper([Status]), 'CANCEL') > 0 OR Index(Upper([Status]), 'REJECTED') > 0 OR [Parsed Status Block] = 'Cancelled', Null(), 
        IF(Left([Parsed Status Block], 8) = 'Received', 
            IF(Match([Parsed Status Block], 'Received (MSM)', 'Received (FUM)'), [Order Date], [Actual Status Date]),
            RangeMax(Today(), [Estimated Arrival Date]) // Forces past-due items to Today
        )
    ) AS [Unified Action Date],

    (IF(Index(Upper([Status]), 'CANCEL') > 0 OR Index(Upper([Status]), 'REJECTED') > 0 OR [Parsed Status Block] = 'Cancelled', Null(), 
        IF(Left([Parsed Status Block], 8) = 'Received', 
            IF(Match([Parsed Status Block], 'Received (MSM)', 'Received (FUM)'), [Order Date], [Actual Status Date]),
            RangeMax(Today(), [Estimated Arrival Date])
        )
    )) - Date([SA00 Date] - 600) AS [Receipt_EAD Relative Days from A-600];

// DOUBLE PRECEDING LOAD
LOAD 
    PROJ_ID, JML_STATUS_CD, OWNER_POSITION_CD, [Project Name],
    PROJ_ID & ' - ' & Trim(Replace([Project Name], 'YO -', '')) AS [Project Display Name],
    IF(ApplyMap('ProjectCohort_Map', PROJ_ID, 'Unknown') = 'Active', PROJ_ID & ' - ' & Trim(Replace([Project Name], 'YO -', '')), Null()) AS [Active Project Filter],
    ApplyMap('ProjectType_Map', PROJ_ID, 'Unknown') AS [Project Type],
    ApplyMap('ProjectCohort_Map', PROJ_ID, 'Unknown') AS [Cohort Type],
    ApplyMap('Map_CA00_Actual', PROJ_ID, Null()) AS [CA00 Date],
    [ICN KO], [JML Doc #], [NIIN or Part Number], [Part Name], [EDD], [COG Code], [ProjectID ICNKO Key], [SKED Start], [Total DU], [Resolved Key Event ID], [Event Name], [KEY EVENT Date], CU_PHASE_SA_ID, [System], [Material Type], [Key Event], [Job Summary Title], [SA00 Date],
    Date([SA00 Date] - 600) AS [A-600 Date], 
    [QTY Due], [NEED Date], [Unit of Issue], [Document Cut Number], [Status], [Parsed Status Block], [Parsed ESD], [Parsed KTR], [Actual Status Date], 
    
    Date(MakeDate( IF(Num(Left([JML Doc #], 1)) > $(vCurrentYearLastDigit), $(vCenturyAndDecade) - 10, $(vCenturyAndDecade)) + Num(Left([JML Doc #], 1)) , 1, 1) + Num(Mid([JML Doc #], 2, 3)) - 1) AS [Order Date], 
    Date(MakeDate( IF(Num(Left([JML Doc #], 1)) > $(vCurrentYearLastDigit), $(vCenturyAndDecade) - 10, $(vCenturyAndDecade)) + Num(Left([JML Doc #], 1)) , 1, 1) + Num(Mid([JML Doc #], 2, 3)) - 1) - Date([SA00 Date] - 600) AS [Days from A-600],
    RangeMax(0, Ceil((Date(MakeDate( IF(Num(Left([JML Doc #], 1)) > $(vCurrentYearLastDigit), $(vCenturyAndDecade) - 10, $(vCenturyAndDecade)) + Num(Left([JML Doc #], 1)) , 1, 1) + Num(Mid([JML Doc #], 2, 3)) - 1) - Date([SA00 Date] - 600)) / 7)) AS [Relative Week from A-600],
    
    IF(Index(Upper([Status]), 'CANCEL') > 0 OR Index(Upper([Status]), 'REJECTED') > 0 OR [Parsed Status Block] = 'Cancelled', 'Cancelled', 
        IF(Left([Parsed Status Block], 8) = 'Received', 'Received', 
            IF(WildMatch([JML Doc #], 'PENDING_ACT_*'), 'In Process', 
                IF(Len(Trim([QTY Due])) = 0, 'Outstanding', IF([QTY Due] <= 0, 'Received', 'Outstanding'))))) AS [Report Status], 
    
    Date([SA00 Date] - 30) AS [A-30 Date], 
    IF(IsNum([SKED Start]),
        IF(IsNum([SA00 Date]), IF([SKED Start] > [SA00 Date], Date([SKED Start] - 14), Date([SA00 Date])), Date([SKED Start] - 14)),
        IF(IsNum([KEY EVENT Date]) AND IsNum([Total DU]), Date( ([KEY EVENT Date] - 14) - (Floor([Total DU] / 6) * 7) - Mod([Total DU], 6) - IF(WeekDay([KEY EVENT Date] - 14) - Mod([Total DU], 6) < 0, 1, 0) ), Date([KEY EVENT Date] - 14))
    ) AS [Project Need Date],
    Coalesce(IF(NOT (Index(Upper([Status]), 'CANCEL') > 0 OR Index(Upper([Status]), 'REJECTED') > 0 OR Left([Parsed Status Block], 8) = 'Received'), Date([EDD] + 14)), IF(IsNum([Parsed ESD]), Date([Parsed ESD] + 14))) AS [Estimated Arrival Date]
RESIDENT FinalReportData WHERE NOT Index(Upper([Status]), 'XFERRED') > 0;

DROP TABLE FinalReportData;

// CANONICAL DATE MODEL & MASTER CALENDAR
// =========================================================================================

Canonical_Link:
// 1. Link Orders to the Master Timeline
LOAD 
    [JML Doc #], 
    [Order Date] AS [Master Date], 
    [Relative Week from A-600] AS [Master Relative Week],
    'Order' AS [Date Type] 
RESIDENT Final_Table 
WHERE IsNum([Order Date]);

// 2. Link Actual Receipts & Projections (EADs) to the Master Timeline
CONCATENATE(Canonical_Link)
LOAD 
    [JML Doc #], 
    [Unified Action Date] AS [Master Date], 
    [Receipt_EAD Relative Week from A-600] AS [Master Relative Week],
    'Action' AS [Date Type] 
RESIDENT Final_Table 
WHERE IsNum([Unified Action Date]);

// -----------------------------------------------------------------------------------------
// MASTER CALENDAR GENERATION
// -----------------------------------------------------------------------------------------
Temp_Bounds:
LOAD 
    Min([Master Relative Week]) AS MinWeek, 
    Max([Master Relative Week]) AS MaxWeek 
RESIDENT Canonical_Link;

LET vMinWeek = Peek('MinWeek', 0, 'Temp_Bounds');
LET vMaxWeek = Peek('MaxWeek', 0, 'Temp_Bounds');

DROP TABLE Temp_Bounds;

Master_Relative_Calendar:
LOAD
    ($(vMinWeek) + RowNo() - 1) AS [Master Relative Week]
AUTOGENERATE ($(vMaxWeek) - $(vMinWeek) + 1);
