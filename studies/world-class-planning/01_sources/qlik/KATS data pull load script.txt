Master Calendar:
//$(Must_Include=lib://AttachedFiles/master_calendar.txt);

//** Creates master calendar table**//

//** Retrieve Min and Max dates from STS field**//

// To create the numerical values, to be used in date math

Min_Max_Temp: 
Load 
    Floor(WeekStart(MIN(KEMS_Desired_DT)))-7 as MinDate,
	Ceil(WeekStart(MAX(KEMS_Desired_DT)))+7 as MaxDate
Resident KEMS
Where Type = 'KE' and (KEMS_ID = 'SA00' or KEMS_ID = 'CA00');	

//*** Generate Min and Max variables

LET vMinDate = PEEK('MinDate', 0, Min_Max_Temp);
LET vMaxDate = PEEK('MaxDate', 0, Min_Max_Temp);

DROP Table Min_Max_Temp;

TRACE vMinDate = $(vMinDate);
TRACE vMaxDate = $(vMaxDate);


MasterCalendarTemp:
Load
	Date($(vMinDate) + RowNo() -1) AS CanonicalDate
	AutoGenerate $(vMaxDate) - $(vMinDate) +1;
 
//**** Create the Master Calendar ****

MasterCalendar:
LOAD
	CanonicalDate,
	Week(CanonicalDate) AS Week,
	Year(CanonicalDate) AS Year,
	Month(CanonicalDate) AS Month,
	Day(CanonicalDate) AS Day,
	WeekDay(CanonicalDate) AS WeekDay,
	Dual('Q' & Ceil(Month(CanonicalDate) / 3), Ceil(Month(CanonicalDate) / 3)) AS Quarter,
	Date(MonthStart(CanonicalDate),'MMM-YYYY') as MonthYear,
	Week(CanonicalDate) & '-' & If(Week(CanonicalDate)=1 AND Month(CanonicalDate)=12,Year (CanonicalDate)+1,
		If(Match(Week(CanonicalDate), 52, 53) AND Month(CanonicalDate)=1, Year(CanonicalDate)-1, Year(CanonicalDate))) AS WeekYear,
    If(Week(CanonicalDate)=1 AND Month(CanonicalDate)=12,Year (CanonicalDate)+1,
		If(Match(Week(CanonicalDate), 52, 53) AND Month(CanonicalDate)=1, Year(CanonicalDate)-1, Year(CanonicalDate))) & '-' & Week(CanonicalDate) AS YearWeek,
	InYearToDate(CanonicalDate, Today(), 0) * -1 AS CYTDFlag,
	InYearToDate(CanonicalDate, Today(), -1) * -1 as LYTDFlag,
	If(DayNumberOfYear(CanonicalDate) <= DayNumberOfYear(Today()), 1, 0) AS IsInYTD,
	If(DayNumberOfQuarter(CanonicalDate) <= DayNumberOfQuarter (Today()), 1, 0) AS IsInQTD,
	If(Day(CanonicalDate) <= Day(Today()), 1, 0) AS IsInMTD,
	If(Month(CanonicalDate) = Month(Today()), 1, 0) AS IsCurrentMonth,
	If(Month(AddMonths(CanonicalDate,1)) = Month(Today()), 1, 0) AS IsLastMonth,
    WeekStart(CanonicalDate) as WK_Start
RESIDENT MasterCalendarTemp
Order by CanonicalDate ASC;

DROP Table MasterCalendarTemp;
